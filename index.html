        function isValidChecksumPin(code) {
            if (!/^[1-9]\d{5}$/.test(code)) return false;
            const d = code.split('').map(Number);
            return d[5] === ((d[0] + d[1] + d[2] + d[3] + d[4]) % 10);
        }

        function generateChecksumPin() {
            let d1, d2, d3, d4, d5;
            do {
                d1 = Math.floor(Math.random() * 9) + 1;
                d2 = Math.floor(Math.random() * 10);
                d3 = Math.floor(Math.random() * 10);
                d4 = Math.floor(Math.random() * 10);
                d5 = Math.floor(Math.random() * 10);
            } while (
                (d1 === d2 && d2 === d3) ||
                (d2 === d3 && d3 === d4) ||
                (d3 === d4 && d4 === d5) ||
                (d2 === d1 + 1 && d3 === d2 + 1 && d4 === d3 + 1)
            );
            const d6 = (d1 + d2 + d3 + d4 + d5) % 10;
            return `${d1}${d2}${d3}${d4}${d5}${d6}`;
        }

        function setupPinInputs() {
            const boxes = document.querySelectorAll('.pin-box');
            boxes.forEach((box, idx) => {
                box.addEventListener('input', (e) => {
                    e.target.value = e.target.value.replace(/[^0-9]/g, '');
                    document.getElementById('pinErrorMsg').classList.add('hidden');
                    if (e.target.value.length === 1 && idx < boxes.length - 1) {
                        boxes[idx + 1].focus();
                    }
                });
                box.addEventListener('keydown', (e) => {
                    if (e.key === 'Backspace' && !e.target.value && idx > 0) {
                        boxes[idx - 1].focus();
                    }
                });
                box.addEventListener('paste', (e) => {
                    e.preventDefault();
                    const pasted = (e.clipboardData || window.clipboardData).getData('text').replace(/[^0-9]/g, '').slice(0, 6);
                    pasted.split('').forEach((char, i) => {
                        if (boxes[i]) boxes[i].value = char;
                    });
                    if (pasted.length > 0) {
                        boxes[Math.min(pasted.length, boxes.length - 1)].focus();
                    }
                });
            });

            document.getElementById('pinForm').addEventListener('submit', (e) => {
                e.preventDefault();
                const code = Array.from(boxes).map(b => b.value).join('');
                if (isValidChecksumPin(code)) {
                    unlockApp();
                } else {
                    document.getElementById('pinErrorMsg').classList.remove('hidden');
                    boxes.forEach(b => b.classList.add('border-rose-400', 'bg-rose-50'));
                    setTimeout(() => boxes.forEach(b => b.classList.remove('border-rose-400', 'bg-rose-50')), 800);
                }
            });

            document.getElementById('adminPinForm').addEventListener('submit', (e) => {
                e.preventDefault();
                const pinVal = String(document.getElementById('adminPinInput').value || '').trim();
                if (pinVal === OWNER_ADMIN_PIN) {
                    closeAdminPinModal();
                    openAdminModal();
                } else {
                    document.getElementById('adminPinError').classList.remove('hidden');
                }
            });
        }

        function randomizeSmartScenario() {
            let pool = SCENARIO_THEMES;
            if (state.selectedSubject !== 'all') {
                const matched = SCENARIO_THEMES.filter(s => s.subjects.includes(state.selectedSubject));
                if (matched.length > 0) pool = matched;
            } else if (state.selectedArea !== 'all') {
                const areaObj = LEARNING_AREAS.find(a => a.id === state.selectedArea);
                if (areaObj) {
                    const matched = SCENARIO_THEMES.filter(s => s.subjects.some(sub => areaObj.subjectIds.includes(sub)));
                    if (matched.length > 0) pool = matched;
                }
            }
            const currentId = state.selectedScenario;
            const candidates = pool.length > 1 ? pool.filter(s => s.id !== currentId) : pool;
            const picked = candidates[Math.floor(Math.random() * candidates.length)];

            state.selectedScenario = picked.id;
            document.getElementById('scenarioDropdown').value = picked.id;
            document.getElementById('customScenarioBox').classList.add('hidden');
            updateContextBadge();
            renderCards();
            showToast('สุ่มธีมสถานการณ์ให้สอดคล้องกับวิชาแล้ว!', picked.name);
        }

        function applyFilters() {
            const areaObj = LEARNING_AREAS.find(a => a.id === state.selectedArea) || LEARNING_AREAS[0];
            const q = state.searchQuery.toLowerCase();

            FILTERED_PROMPTS = ALL_PROMPTS.filter(item => {
                if (state.selectedGrade !== 'ทุกระดับชั้น' && item.grade !== state.selectedGrade) return false;
                if (state.selectedArea !== 'all' && !areaObj.subjectIds.includes(item.subjectId)) return false;
                if (state.selectedSubject !== 'all' && item.subjectId !== state.selectedSubject) return false;
                if (q) {
                    const hay = `${item.topic} ${item.subjectName} ${item.grade}`.toLowerCase();
                    if (!hay.includes(q)) return false;
                }
                return true;
            });

            // ระบบสร้างการ์ดหัวข้อใหม่ทันใจ (Custom Topic Instant Card) เมื่อพิมพ์ค้นหาเรื่องใหม่
            CUSTOM_TOPIC_ITEM = null;
            if (state.searchQuery.length > 0) {
                const exactExists = FILTERED_PROMPTS.some(p => p.topic.toLowerCase() === q);
                if (!exactExists) {
                    const targetSubj = state.selectedSubject !== 'all'
                        ? (SUBJECTS.find(s => s.id === state.selectedSubject) || SUBJECTS[0])
                        : (state.selectedArea !== 'all'
                            ? (SUBJECTS.find(s => areaObj.subjectIds.includes(s.id)) || SUBJECTS[0])
                            : SUBJECTS[0]);
                    const targetGrade = state.selectedGrade !== 'ทุกระดับชั้น' ? state.selectedGrade : 'ป.4';

                    CUSTOM_TOPIC_ITEM = {
                        id: 'PC-CUSTOM',
                        isCustom: true,
                        areaId: targetSubj.areaId,
                        subjectId: targetSubj.id,
                        subjectName: targetSubj.name,
                        subjectIcon: targetSubj.icon,
                        grade: targetGrade,
                        topic: state.searchQuery,
                        pageCount: 1,
                        wsActivities: ['matching', 'fill_blanks', 'multiple_choice'],
                        wsQuestionCount: 10,
                        wsStyleIdx: 5,
                        wsIncludeAnswerKey: false,
                        wsThemeFrameCut: false,
                        mdStyleIdx: 0,
                        mdSizeIdx: 4,
                        mdFormatIdx: 0,
                        mdCharIdx: targetSubj.id === 'english' ? 0 : 4
                    };
                }
            }

            displayedCount = PAGE_SIZE;
            const totalFound = FILTERED_PROMPTS.length + (CUSTOM_TOPIC_ITEM ? 1 : 0);
            document.getElementById('filteredCountText').textContent = totalFound.toLocaleString();

            const gradeLabel = state.selectedGrade === 'ทุกระดับชั้น' ? 'ทุกระดับชั้น' : state.selectedGrade;
            const areaLabel = state.selectedArea === 'all' ? 'ทุกกลุ่มสาระ' : areaObj.name.replace(/^\d+\.\s*/, '');
            const subjObj = SUBJECTS.find(s => s.id === state.selectedSubject);
            const subjLabel = state.selectedSubject === 'all' ? 'ทุกรายวิชา' : subjObj.name;
            document.getElementById('activeFilterSummary').textContent = `${gradeLabel} · ${areaLabel} · ${subjLabel}`;

            renderCards();
        }

        function buildContextDirectivesBlock() {
            const blocks = [];

            if (state.extraLesson.length > 0) {
                blocks.push(`[เนื้อหาบทเรียนและข้อมูลเฉพาะที่ครูผู้สอนกำหนด (บังคับใช้เป็นข้อมูลหลัก)]\n${state.extraLesson}\nคำสั่งกำกับ: ให้นำเนื้อหา คำศัพท์ หรือข้อมูลข้างต้นมาใช้เป็นแกนหลักในการสร้างเนื้อหาและโจทย์อย่างครบถ้วนถูกต้อง`);
            }

            let scenarioDetail = '';
            let scenarioLabel = '';
            if (state.selectedScenario === 'custom' && state.customScenarioText.length > 0) {
                scenarioLabel = 'สถานการณ์เฉพาะที่ครูผู้สอนกำหนด';
                scenarioDetail = state.customScenarioText;
            } else if (state.selectedScenario !== 'none' && state.selectedScenario !== 'custom') {
                const scObj = SCENARIO_THEMES.find(s => s.id === state.selectedScenario);
                if (scObj) {
                    scenarioLabel = scObj.name;
                    scenarioDetail = scObj.detail;
                }
            }

            const scopeObj = state.selectedSocialScope !== 'none'
                ? SOCIAL_SCOPES.find(sc => sc.id === state.selectedSocialScope)
                : null;

            if (scenarioDetail || scopeObj) {
                let scText = `[สถานการณ์จำลองและบริบทวงสังคม (Scenario & Social Context)]\n`;
                if (scenarioDetail) {
                    scText += `- ธีมสถานการณ์ที่อยู่ในความสนใจ (${scenarioLabel}): ${scenarioDetail}\n`;
                }
                if (scopeObj) {
                    scText += `- ขอบเขตบริบทวงสังคม (${scopeObj.name}): ${scopeObj.detail}\n`;
                }
                scText += `- กฎการออกแบบสถานการณ์: ผูกเนื้อหาบทเรียนเข้ากับสถานการณ์และบริบทวงสังคมข้างต้นอย่างเป็นธรรมชาติ สรุปคำบรรยายสถานการณ์ให้กระชับชัดเจน (2–3 บรรทัด ไม่เกิน 35 คำ เพื่อไม่ให้ตัวหนังสือล้นหน้ากระดาษ) พร้อมภาพประกอบเหตุการณ์ที่สอดคล้องกับช่วงวัยของผู้เรียน`;
                blocks.push(scText);
            }

            return blocks.length > 0 ? `\n\n${blocks.join('\n\n')}` : '';
        }

        function toggleCardActivity(id, actId, checkboxEl) {
            const item = getCardById(id);
            if (!item) return;
            if (checkboxEl.checked) {
                if (item.wsActivities.length >= 6) {
                    checkboxEl.checked = false;
                    showToast('เลือกได้สูงสุด 6 กิจกรรมต่อ 1 ใบงาน', 'กรุณายกเลิกบางกิจกรรมก่อนเลือกเพิ่มครับ');
                    return;
                }
                if (!item.wsActivities.includes(actId)) item.wsActivities.push(actId);
            } else {
                if (item.wsActivities.length <= 1) {
                    checkboxEl.checked = true;
                    showToast('ต้องเลือกอย่างน้อย 1 กิจกรรม', 'ใบงานจำเป็นต้องมีกิจกรรมอย่างน้อย 1 รูปแบบครับ');
                    return;
                }
                item.wsActivities = item.wsActivities.filter(x => x !== actId);
            }
            renderCards();
        }

        }

        function copyTextToClipboard(text, title, subtitle) {
            const fallbackCopy = () => {
                const ta = document.createElement('textarea');
                ta.value = text;
                ta.setAttribute('readonly', '');
                ta.style.position = 'fixed';
                ta.style.left = '-9999px';
                ta.style.top = '0';
                document.body.appendChild(ta);
                ta.focus();
                ta.select();
                try {
                    document.execCommand('copy');
                } catch (err) {}
                document.body.removeChild(ta);
                showToast(title, subtitle);
            };

            if (navigator.clipboard && window.isSecureContext) {
                navigator.clipboard.writeText(text)
                    .then(() => showToast(title, subtitle))
                    .catch(() => fallbackCopy());
            } else {
                fallbackCopy();
            }
        }

        let toastTimer = null;
        function showToast(title, subtitle) {
            const toast = document.getElementById('toastNotification');
            document.getElementById('toastTitle').textContent = title;
            document.getElementById('toastSub').textContent = subtitle;
            toast.classList.remove('translate-y-24', 'opacity-0');
            clearTimeout(toastTimer);
            toastTimer = setTimeout(() => {
                toast.classList.add('translate-y-24', 'opacity-0');
            }, 3000);
        }
